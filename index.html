# Fallback: create a rough MIDI by manual pitch sampling using librosa without pyin import librosa import numpy as np from mido import MidiFile, MidiTrack, Message audio_path = "/mnt/data/10月30日 下午6点01分.mp3" y, sr = librosa.load(audio_path, mono=True) # Use spectral centroid as rough pitch proxy (very rough but works) S = np.abs(librosa.stft(y)) freqs = librosa.fft_frequencies(sr=sr) dominant_freqs = freqs[np.argmax(S, axis=0)] times = librosa.frames_to_time(np.arange(len(dominant_freqs)), sr=sr) notes = [] current_note = None start_time = None for f, t in zip(dominant_freqs, times): if f > 50: midi = int(round(librosa.hz_to_midi(f))) if current_note is None: current_note = midi start_time = t elif midi != current_note: notes.append((current_note, start_time, t)) current_note = midi start_time = t if current_note is not None: notes.append((current_note, start_time, times[-1])) mid = MidiFile() track = MidiTrack() mid.tracks.append(track) track.append(Message('program_change', program=40, time=0)) ticks_per_beat = mid.ticks_per_beat def sec_to_ticks(sec): return int(sec * ticks_per_beat * 2) last_tick = 0 for note, start, end in notes: start_tick = sec_to_ticks(start) end_tick = sec_to_ticks(end) delta = max(0, start_tick - last_tick) track.append(Message('note_on', note=note, velocity=64, time=delta)) track.append(Message('note_off', note=note, velocity=64, time=max(1, end_tick - start_tick))) last_tick = end_tick midi_path = "/mnt/data/rough_violin.mid" mid.save(midi_path) midi_path
